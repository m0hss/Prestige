{{- /* Orders case studies by weight (default 100) ascending, ties by date descending (3.4). Context: a slice of case-study pages. */ -}} {{- $rows := slice -}} {{- range . -}} {{- $w := .Params.weight | default 100 -}} {{- $rows = $rows | append (dict "k" (printf "%010d-%012d" (add (int $w) 1000000000) (sub 999999999999 .Date.Unix)) "p" .) -}} {{- end -}} {{- $out := slice -}} {{- range sort $rows "k" }}{{ $out = $out | append .p }}{{ end -}} {{- return $out -}}
